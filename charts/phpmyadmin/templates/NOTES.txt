2. Service Access
{{- if .Values.ingress.enabled }}
   Ingress is enabled. Access phpMyAdmin at:
  {{- range .Values.ingress.hosts }}
   - http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}{{ (index .paths 0).path }}
  {{- end }}
{{- else if contains "NodePort" .Values.service.type }}
   NodePort is enabled. Retrieve the address with:
   $ export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "phpmyadmin.fullname" . }})
   $ export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   $ echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
   LoadBalancer is enabled. Retrieve the external IP with:
   $ kubectl get svc --namespace {{ .Release.Namespace }} {{ include "phpmyadmin.fullname" . }} -w
{{- else }}
   Port-forward the service to localhost:
   $ kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "phpmyadmin.fullname" . }} 8080:{{ .Values.service.port }}
   Then browse to: http://127.0.0.1:8080
{{- end }}
