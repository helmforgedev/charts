  {{- range .Values.ingress.hosts }}

  {{- end }}

  {{- range .Values.gatewayAPI.httpRoutes }}
    {{- range .hostnames }}
   - http://{{ . }}/
    {{- end }}
  {{- end }}
{{- else }}
   Port-forward the Transmute service to localhost:
   $ kubectl port-forward svc/{{ include "transmute.fullname" . }} 8080:{{ .Values.service.port }} -n {{ .Release.Namespace }}
   Then browse to: http://127.0.0.1:8080/
{{- end }}
